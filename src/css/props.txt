# CSS property database. Input of tools/cssgen, which writes src/css/props.inc.
#
#   name | kind | inherited | keywords | types | initial | anim | interp | group | longhands [| logic]
#
#   kind        L longhand, S shorthand
#   inherited   0 or 1
#   keywords    comma-separated keywords accepted as the whole value, or -
#   types       value types: len pct lenpct num int pint (integer >= 1) alpha angle time res color url str ident list,
#               g:<grammar> (compound value),
#               or -. A '+' makes numeric types non-negative. 'list' = comma-separated list.
#   initial     CSS text, parsed on demand. - for shorthands.
#   Known gaps (discrete although the spec interpolates): clip (rect), grid-template-columns/rows (track lists), font-variation-settings (tag/number list).
#   anim        animation type: N not animatable, D discrete, L length-percentage, F number, C color,
#               X compound (the interp kind decides how the value mixes; see src/anim/interp.c)
#   interp      interpolation kind of the computed value's type: - (discrete), len (length, percentage,
#               calc, or number pairs), num (number or percentage), int (integer, rounded), color, shadow,
#               filter, xform (transform list), shape (basic shape), pos (two-axis position), axis
#               (background-position-x and -y lists), lplist (lists of lengths and numbers), vis (visibility),
#               disp (display), cv (content-visibility), ratio (<ratio>)
#   group       style struct group for W2d, or -
#   longhands   shorthand components, or -
#
#   logic       optional (11th field): P:<kind>.<family>:<role> physical or L:<kind>.<family>:<role>
#               flow-relative, e.g. P:box.margin:left and L:box.margin:inline-start. Properties of one
#               family map onto each other per writing mode (embi_prop_to_physical in props.h).
#
# Grammar scope: simple values are typed by the 'types' field. Compound values use 'g:<name>',
# which selects the function embi_gram_<name> (src/css/value_*.c): it parses the whole value, or
# each comma-separated item when 'list' is also given, into a typed structure (props.h). Keywords
# listed for a property are the ones valid as the entire value (they parse to a KEYWORD before the
# grammar runs); keywords that only occur inside a compound value are declared with @kw. Every
# value is validated and typed; there is no unparsed fallback. Shorthands are expanded by
# src/css/shorthand.c. Adding a property is one line here plus a regeneration
# (cmake --build build/<dir> --target cssgen).
#
# Extra keywords used inside compound grammars (not valid as a whole value of any property).
